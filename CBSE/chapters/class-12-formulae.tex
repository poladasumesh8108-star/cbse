\item area of a triangle OAB 
\begin{align}
    Ar(\triangle OAB)= \frac{1}{2}\vert{}\overrightarrow{OA}\times \overrightarrow{OB}\vert{} \label{eq:ee26btech11222-prob 1}
\end{align}
\item 
\begin{align}
    sin^{-1}x+cos^{-1}x=\frac{\pi}{2} \label{eq:ee26btech11222-prob 2}
\end{align}
\item angle between vectors
\begin{align}
    cos\theta=\left(\frac{\overrightarrow{u}\cdot\overrightarrow{v}}{\vert{\overrightarrow{u}\vert{}\vert{\overrightarrow{v}\vert{}}}}\right) \label{eq:ee26btech11222-prob 8}
\end{align}

\item $A\epsilon R^{n\times n}$
\begin{align} 
    \vert{A}\vert{}=\vert{kI}\vert{} \implies \vert{A}\vert{}= k^n\vert{I}\vert{} \label{eq:ee26btech11222-prob 9}
\end{align}

\item 
\begin{align}
    cos^{-1}(-x)=\pi - cos^{-1}x \label{eq:ee26btech11222-prob 14}
\end{align}
 \item distance between parallel planes $ax+by+cz=d_1$ and $ax+by+cz=d_2$
 \begin{align}
     d=\left | \frac{d_2-d_1}{\sqrt{a^2+b^2+c^2}}\right |
 \end{align}
 \item king's property
 \begin{align}
     \int_a^b f(x)dx= \int_a^b f(a+b-x)dx \label{eq:ee26btech11222-prob 16}
 \end{align}
 \item condition for a function to be continuous at x=a
 \begin{align}
     f(a^-)=f(a^+) \label{eq:ee26btech11222-prob 18}
 \end{align}
 \item y is a function of x
 \begin{align}
     \frac{dy}{dx}+py=q \\ \implies ye^{\int pdx}= \int q  e^{\int pdx}dx \label{eq:ee26btech11222-prob 19}
 \end{align} \newpage 
 \item differentiation of a function$f(x)$ with respect to other function $g(x)$
 \begin{align}
     \frac{d(f(x))}{d(g(x))}=\frac{f^\prime (x)}{g^\prime (x)} \label{eq:ee26btech11222-prob 20}
 \end{align}
 \item $\alpha ,\beta, \gamma $ are the angles made by vector with x,y,z axes respectively 
 \begin{align} 
     cos^2\alpha+ cos^2\beta+cos^2\gamma=1 \label{eq:ee26btech11222-prob 21}
 \end{align}
 \item characteristic polynomial of a matrix A
 \begin{align}
    |A-\lambda I|=0 \label{eq:ee26btech11222-prob 22} 
 \end{align}
 \item integration by parts 
 \begin{align}
     \int f(x)g(x)dx=f(x)\int g(x)dx-\int f^\prime (x)\left(\int g(x)dx\right) dx \label{eq:ee26btech11222-prob 29}
 \end{align}
 \item inverse of a matrix A
 \begin{align}
     A^{-1}= \frac{1}{\vert{}A\vert{}}adj(A) \label{eq:ee26btech11222-prob }
 \end{align}

 \item condition for minimum surface area 
 \begin{align}
     S=\pi r^2+2\pi rh \implies \frac{dS}{dr}=0 \\ \implies \frac{d^2S}{dr^2}>0 \label{eq:ee26btech11222-prob 33}
 \end{align}
 \item 
 \begin{align}
     \vert{}\overrightarrow{a} \overrightarrow{b} \overrightarrow{c_1}+\overrightarrow{c_2}\vert{}=|\overrightarrow{a} \overrightarrow{b} \overrightarrow{c_1}|+|\overrightarrow{a} \overrightarrow{b} \overrightarrow{c_2}|
 \end{align}
 \item Bayes' theorem 
 \begin{align}
     P(A|B)=\frac{P(B|A)P(A)}{P(B)} \label{eq:ee26btech11222-prob 36}
 \end{align}
\end{enumerate}
